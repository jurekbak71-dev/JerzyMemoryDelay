cmake_minimum_required(VERSION 3.25)
project(JerzyMemoryDelay VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(JERZY_BUILD_PLUGIN "Build the VST3 plugin" ON)
option(JERZY_GUI "Build the hardware-style VSTGUI editor" ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(MemoryDelayTests tests/dsp_tests.cpp)
  target_include_directories(MemoryDelayTests PRIVATE source)
  add_test(NAME MemoryDelayDSP COMMAND MemoryDelayTests)
endif()
if(JERZY_BUILD_PLUGIN)
  set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
  # Add only the editor libraries; the SDK convenience path also forces its
  # unrelated standalone application framework (and GTK) to build.
  set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
  set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
  set(VSTGUI_STANDALONE OFF CACHE BOOL "" FORCE)
  set(VSTGUI_TOOLS OFF CACHE BOOL "" FORCE)
  set(VST3_SDK_ROOT "" CACHE PATH "Existing recursive VST3 SDK checkout")
  if(VST3_SDK_ROOT)
    set(JERZY_SDK_DIR "${VST3_SDK_ROOT}")
    add_subdirectory("${VST3_SDK_ROOT}" "${CMAKE_BINARY_DIR}/vst3sdk")
  else()
    include(FetchContent)
    FetchContent_Declare(vst3sdk GIT_REPOSITORY https://github.com/steinbergmedia/vst3sdk.git
      GIT_TAG 3cdf9ca5d1f5b1b21e0a86832aa4abe55607bd96 GIT_SUBMODULES_RECURSE TRUE)
    FetchContent_MakeAvailable(vst3sdk)
    set(JERZY_SDK_DIR "${vst3sdk_SOURCE_DIR}")
  endif()
  if(JERZY_GUI)
    set(public_sdk_SOURCE_DIR "${JERZY_SDK_DIR}/public.sdk")
    macro(smtg_add_vstgui_subdirectory gui_dir)
      set(VSTGUI_DISABLE_UNITTESTS 1)
      set(VSTGUI_STANDALONE OFF)
      set(VSTGUI_STANDALONE_EXAMPLES OFF)
      set(VSTGUI_TOOLS OFF)
      add_subdirectory("${gui_dir}" "${CMAKE_CURRENT_BINARY_DIR}/VSTGUI.build")
    endmacro()
    smtg_enable_vstgui_support(VSTGUI_SOURCE_DIR "${JERZY_SDK_DIR}/vstgui4")
  endif()
  smtg_enable_vst3_sdk()
  smtg_add_vst3plugin(JerzyMemoryDelay source/plugin.cpp source/dsp.h source/parameters.h)
  target_link_libraries(JerzyMemoryDelay PRIVATE sdk)
  target_compile_definitions(JerzyMemoryDelay PRIVATE JERZY_GUI=$<BOOL:${JERZY_GUI}>)
  if(JERZY_GUI)
    target_sources(JerzyMemoryDelay PRIVATE source/editor.cpp)
    target_link_libraries(JerzyMemoryDelay PRIVATE vstgui_support)
    smtg_target_add_plugin_resources(JerzyMemoryDelay RESOURCES resource/memory.uidesc)
  endif()
  if(SMTG_MAC)
    smtg_target_set_bundle(JerzyMemoryDelay BUNDLE_IDENTIFIER "pl.jerzyaudio.memorydelay" COMPANY_NAME "Jerzy Audio")
  endif()
  if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    if(WIN32)
      set(module_source "${JERZY_SDK_DIR}/public.sdk/source/vst/hosting/module_win32.cpp")
    elseif(UNIX AND NOT APPLE)
      set(module_source "${JERZY_SDK_DIR}/public.sdk/source/vst/hosting/module_linux.cpp")
    endif()
    if(module_source)
      add_executable(MemoryDelayHostTests tests/host_tests.cpp "${module_source}"
        "${JERZY_SDK_DIR}/public.sdk/source/common/memorystream.cpp")
      target_include_directories(MemoryDelayHostTests PRIVATE source)
      target_link_libraries(MemoryDelayHostTests PRIVATE sdk_hosting)
      add_dependencies(MemoryDelayHostTests JerzyMemoryDelay)
      add_test(NAME MemoryDelayHost COMMAND MemoryDelayHostTests $<TARGET_FILE_DIR:JerzyMemoryDelay>/../..)
      if(JERZY_GUI AND UNIX AND NOT APPLE)
        find_package(X11 REQUIRED)
        add_executable(MemoryDelayGUI tests/gui_smoke.cpp "${module_source}")
        target_include_directories(MemoryDelayGUI PRIVATE source ${X11_INCLUDE_DIR})
        target_link_libraries(MemoryDelayGUI PRIVATE sdk_hosting ${X11_LIBRARIES})
        add_dependencies(MemoryDelayGUI JerzyMemoryDelay)
        add_test(NAME MemoryDelayGUI COMMAND MemoryDelayGUI $<TARGET_FILE_DIR:JerzyMemoryDelay>/../.. "${CMAKE_BINARY_DIR}/memory-delay.ppm")
        set_tests_properties(MemoryDelayGUI PROPERTIES TIMEOUT 30)
      endif()
    endif()
  endif()
endif()
