cmake_minimum_required(VERSION 3.25)
project(JerzyMultiFXNext VERSION 0.1.0 LANGUAGES C CXX)

include(CTest)
option(NEXT_JACK "Build JACK2 backend" ON)
set(NEXT_SANITIZER "" CACHE STRING "address,undefined or thread")
if(NEXT_SANITIZER)
 add_compile_definitions(NEXT_SANITIZED=1)
 add_compile_options(-fsanitize=${NEXT_SANITIZER} -fno-omit-frame-pointer -g)
 add_link_options(-fsanitize=${NEXT_SANITIZER})
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
add_library(next_core src/plugin.cpp src/engine.cpp src/session.cpp)
target_include_directories(next_core PUBLIC include src vendor)
target_link_libraries(next_core PUBLIC ${CMAKE_DL_LIBS})
add_library(next_pffft STATIC vendor/pffft/src/pffft.c vendor/pffft/src/pffft_common.c)
target_include_directories(next_pffft PUBLIC vendor/pffft/include/pffft)
target_compile_definitions(next_pffft PUBLIC PFFFT_STATIC_DEFINE=1)
set_target_properties(next_pffft PROPERTIES C_VISIBILITY_PRESET hidden)
add_library(next_cab_dsp STATIC src/pffft_cab.cpp)
target_include_directories(next_cab_dsp PUBLIC src)
target_link_libraries(next_cab_dsp PUBLIC next_pffft)
configure_file(vendor/pffft/NOTICES.txt "${CMAKE_BINARY_DIR}/plugins/cab/assets/licenses/PFFFT-NOTICES.txt" COPYONLY)
configure_file(vendor/pffft/LICENSE.txt "${CMAKE_BINARY_DIR}/plugins/cab/assets/licenses/PFFFT.txt" COPYONLY)
foreach(p gain synth cab legacy)
 configure_file(plugins/${p}.json "${CMAKE_BINARY_DIR}/plugins/${p}/manifest.json" COPYONLY)
 add_library(next_${p} MODULE plugins/${p}.cpp)
 target_include_directories(next_${p} PRIVATE include)
 target_link_libraries(next_${p} PRIVATE ${CMAKE_DL_LIBS})
 set_target_properties(next_${p} PROPERTIES PREFIX "" OUTPUT_NAME plugin LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/${p}")
endforeach()
target_link_libraries(next_cab PRIVATE next_cab_dsp)
if(BUILD_TESTING)
 add_executable(next_convolver_test tests/convolver_test.cpp)

  add_library(${p} MODULE tests/${p}.c)
  target_include_directories(${p} PRIVATE include)
  set_target_properties(${p} PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
 endforeach()

 add_dependencies(next_plugin_test next_gain next_synth next_cab wrong noentry invalid)
 add_executable(next_engine_test tests/engine_test.cpp)
 target_link_libraries(next_engine_test PRIVATE next_core)
 add_dependencies(next_engine_test next_plugin_test)
 add_test(NAME NextEngine COMMAND next_engine_test "${CMAKE_BINARY_DIR}/plugins")

 target_link_libraries(next_session_test PRIVATE next_core)
 add_dependencies(next_session_test next_gain next_synth)
 add_test(NAME NextSession COMMAND next_session_test "${CMAKE_BINARY_DIR}/plugins")

  find_program(DPKG_DEB_EXECUTABLE dpkg-deb)
  find_program(DESKTOP_VALIDATE_EXECUTABLE desktop-file-validate)
  if(DPKG_DEB_EXECUTABLE AND DESKTOP_VALIDATE_EXECUTABLE)
   add_test(NAME NextClickPackages COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/click_package_test.py")
  endif()
  add_test(NAME NextInstallers COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/installer_test.py")
  add_test(NAME NextPackages COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/package_test.py" "${CMAKE_BINARY_DIR}/plugins")

  add_library(next_alloc_guard SHARED tests/rt_alloc_guard.cpp)
  target_link_libraries(next_engine_test PRIVATE next_alloc_guard)
  target_compile_definitions(next_engine_test PRIVATE NEXT_ALLOC_GUARD=1)

  target_link_libraries(next_convolver_failure_test PRIVATE next_cab_dsp next_alloc_guard)

  target_link_libraries(next_convolver_test PRIVATE next_alloc_guard)
  target_compile_definitions(next_convolver_test PRIVATE NEXT_ALLOC_GUARD=1)

endif()
